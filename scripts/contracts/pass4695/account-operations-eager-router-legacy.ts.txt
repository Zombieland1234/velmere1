type OperationKey = keyof typeof OPERATIONS;

function resolveOperation(operation: string) {
  return Object.prototype.hasOwnProperty.call(OPERATIONS, operation)
    ? OPERATIONS[operation as OperationKey]
    : null;
}

export async function GET(request: Request, context: { params: Promise<{ operation: string }> }) {
  const { operation } = await context.params;
  const handler = resolveOperation(operation);
  if (!handler || typeof handler.GET !== "function") {
    return Response.json({ ok: false, error: "unknown_account_operation", operation }, { status: 404 });
  }
  return handler.GET(request);
}

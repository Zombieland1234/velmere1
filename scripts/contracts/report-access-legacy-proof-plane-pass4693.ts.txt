function tierFromRequest(value: string | null): VelmereTier {
  const normalized = (value ?? "Basic").toLowerCase();
  if (normalized === "advanced") return "Advanced";
  if (normalized === "pro") return "Pro";
  return "Basic";
}

function familyFromRequest(value: string | null): VelmereReportAssetFamily {
  const normalized = (value ?? "unknown").toLowerCase();
  if (["native_crypto", "erc20", "stablecoin", "defi_protocol", "exchange_health", "equity", "etf", "fx", "commodity", "real_estate"].includes(normalized)) {
    return normalized as VelmereReportAssetFamily;
  }
  return "unknown";
}

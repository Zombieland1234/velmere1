type Pass4158AdminAiNegativeCase =
  | "ignore_previous_instructions"
  | "payment_or_entitlement_bypass"
  | "secret_or_env_exfiltration"
  | "raw_prompt_echo_request";

type Pass4158AdminAiPromptInjectionNegativeFixture = {
  passId: "PASS4158_ADMIN_AI_PROMPT_INJECTION_NEGATIVE_FIXTURE";
  inputDigest: string;
  inputLength: number;
  rawInputReturned: false;
  providerBlocked: boolean;
  matchedNegativeCases: Pass4158AdminAiNegativeCase[];
  expectedNegativeCases: Pass4158AdminAiNegativeCase[];
  boundary: typeof PASS4158_ADMIN_AI_PROMPT_INJECTION_NEGATIVE_FIXTURE;
};

function detectPass4158AdminAiNegativeCases(
  input: string,
): Pass4158AdminAiNegativeCase[] {
  const normalized = input.toLowerCase();
  const cases = new Set<Pass4158AdminAiNegativeCase>();
  if (
    /ignore|bypass|override|forget|jailbreak|developer mode/i.test(normalized)
  ) {
    cases.add("ignore_previous_instructions");
  }
  if (
    /payment|paid access|entitlement|stripe|checkout|webhook|402/i.test(
      normalized,
    )
  ) {
    cases.add("payment_or_entitlement_bypass");
  }
  if (
    /secret|api key|apikey|env|token|bearer|private key|seed phrase/i.test(
      normalized,
    )
  ) {
    cases.add("secret_or_env_exfiltration");
  }
  if (/repeat.*prompt|show.*system|echo.*input|raw prompt/i.test(normalized)) {
    cases.add("raw_prompt_echo_request");
  }
  return Array.from(cases);
}

function buildPass4158AdminAiPromptInjectionNegativeFixture(
  input: string,
): Pass4158AdminAiPromptInjectionNegativeFixture {
  const matchedNegativeCases = detectPass4158AdminAiNegativeCases(input);
  return {
    passId: "PASS4158_ADMIN_AI_PROMPT_INJECTION_NEGATIVE_FIXTURE",
    inputDigest: createHash("sha256").update(input, "utf8").digest("hex"),
    inputLength: input.length,
    rawInputReturned: false,
    providerBlocked: matchedNegativeCases.length > 0,
    matchedNegativeCases,
    expectedNegativeCases: [
      "ignore_previous_instructions",
      "payment_or_entitlement_bypass",
      "secret_or_env_exfiltration",
      "raw_prompt_echo_request",
    ],
    boundary: PASS4158_ADMIN_AI_PROMPT_INJECTION_NEGATIVE_FIXTURE,
  };
}

type AiRequestBody = {
  task?: "product_description" | "rewrite_copy" | "alt_text" | "import_cleanup";
  locale?: "pl" | "en" | "de";
  input?: string;
};

const PASS4156_ADMIN_AI_ROUTE_BOUNDARY =
  "pass4156-admin-ai-route-boundary: admin AI generation requires same-origin/rate-limit/admin auth and returns only provider payload plus bounded proof metadata" as const;

type Pass4156AdminAiTask = NonNullable<AiRequestBody["task"]>;
type Pass4156AdminAiLocale = NonNullable<AiRequestBody["locale"]>;

type Pass4156AdminAiBoundary = {
  passId: "PASS4156_ADMIN_AI_ROUTE_BOUNDARY";
  adminAuthRequired: true;
  sameOriginChecked: true;
  inputReturnedRaw: false;
  task: Pass4156AdminAiTask;
  locale: Pass4156AdminAiLocale;
  inputLength: number;
  boundary: typeof PASS4156_ADMIN_AI_ROUTE_BOUNDARY;
};

function resolvePass4156AdminAiTask(value: unknown): Pass4156AdminAiTask {
  return value === "product_description" ||
    value === "rewrite_copy" ||
    value === "alt_text" ||
    value === "import_cleanup"
    ? value
    : "import_cleanup";
}

function resolvePass4156AdminAiLocale(value: unknown): Pass4156AdminAiLocale {
  return value === "en" || value === "de" || value === "pl" ? value : "pl";
}

function parsePass4156AdminAiBody(value: unknown): AiRequestBody {
  return value && typeof value === "object" ? (value as AiRequestBody) : {};
}

function buildPass4156AdminAiBoundary(args: {
  task: Pass4156AdminAiTask;
  locale: Pass4156AdminAiLocale;
  inputLength: number;
}): Pass4156AdminAiBoundary {
  return {
    passId: "PASS4156_ADMIN_AI_ROUTE_BOUNDARY",
    adminAuthRequired: true,
    sameOriginChecked: true,
    inputReturnedRaw: false,
    task: args.task,
    locale: args.locale,
    inputLength: args.inputLength,
    boundary: PASS4156_ADMIN_AI_ROUTE_BOUNDARY,
  };
}

async function handleAdminAiPost(req: Request) {
  const pass2177SizeGuard = rejectPass2177LargeContentLength(req, 256 * 1024);
  if (pass2177SizeGuard) return pass2177SizeGuard;

  const pass2177OriginGuard = assertPass2177SameOriginRequest(req, {
    allowMissingOrigin: true,
  });
  if (pass2177OriginGuard) return pass2177OriginGuard;

  const pass2177RateLimit = applyPass2177SoftRateLimit(req, {
    keyPrefix: "pass2177-ai",
    limit: 24,
    windowMs: 60_000,
  });
  if (!pass2177RateLimit.ok) return pass2177RateLimit.response;

  const methodGuard = assertAllowedMethods(req, ["POST"]);
  if (methodGuard) return methodGuard;

  const sizeGuard = rejectLargeContentLength(req, 64 * 1024);
  if (sizeGuard) return sizeGuard;

  const rateLimit = applySoftRateLimit(req, {
    keyPrefix: "admin-ai",
    limit: 20,
    windowMs: 60_000,
  });
  if (!rateLimit.ok) return rateLimit.response;

  const auth = verifyAdminImportRequest(req);
  if (!auth.ok) return auth.response;

  const apiKey = process.env.GEMINI_API_KEY;

  if (!apiKey) {
    return securityJson(
      { error: "ai_provider_unavailable" },
      {
        status: 503,
        headers: {
          "x-velmere-pass4196-secret-exposure-lock":
            "admin-auth-passed-provider-config-not-disclosed",
        },
      },
    );
  }

  const parsedBody = await readBoundedJsonBody<unknown>(req, 64 * 1024, { maxDepth: 10 });
  if (!parsedBody.ok) return parsedBody.response;
  const body = parsePass4156AdminAiBody(parsedBody.value);
  const task = resolvePass4156AdminAiTask(body.task);
  const locale = resolvePass4156AdminAiLocale(body.locale);
  const input = typeof body.input === "string" ? body.input.trim() : "";

  if (!input) {
    return securityJson({ error: "Input is required." }, { status: 400 });
  }

  const pass4158 = buildPass4158AdminAiPromptInjectionNegativeFixture(input);
  if (pass4158.providerBlocked) {
    return securityJson(
      {
        error: "ai_input_policy_blocked",
        pass4157: buildPass4157AdminAiInputRedactionFixture(input),
        pass4158,
      },
      {
        status: 400,
        headers: {
          "x-velmere-pass4158-admin-ai-negative-fixture":
            "blocked-before-provider-digest-only",
          "x-velmere-pass4159-admin-ai-negative-replay":
            PASS4159_ADMIN_AI_NEGATIVE_CASE_REPLAY_RUNNER,
          "x-velmere-pass4160-admin-ai-signed-negative-fixture-binder":
            PASS4160_ADMIN_AI_SIGNED_NEGATIVE_FIXTURE_BINDER,
          "x-velmere-pass4161-admin-ai-final-runner-intake":
            PASS4161_ADMIN_AI_FINAL_RUNNER_INTAKE_GATE,
          "x-velmere-pass4162-admin-ai-node20-preflight-evidence":
            PASS4162_ADMIN_AI_NODE20_PREFLIGHT_EVIDENCE_BUCKET,
          "x-velmere-pass4163-admin-ai-typecheck-diagnostic-reducer":
            PASS4163_ADMIN_AI_TYPECHECK_DIAGNOSTIC_REDUCER,
          "x-velmere-pass4164-admin-ai-dependency-type-surface":
            PASS4164_ADMIN_AI_DEPENDENCY_TYPE_SURFACE_CLOSURE,
          "x-velmere-pass4165-admin-ai-node20-type-package-alignment":
            PASS4165_ADMIN_AI_NODE20_TYPE_PACKAGE_ALIGNMENT,
          "x-velmere-pass4166-admin-ai-clean-room-install-readiness":
            PASS4166_ADMIN_AI_CLEAN_ROOM_INSTALL_READINESS_LOCK,
          "x-velmere-pass4167-admin-ai-final-cleanroom-execution-dag":
            PASS4167_ADMIN_AI_FINAL_CLEANROOM_EXECUTION_DAG,
          "x-velmere-pass4168-admin-ai-zero-skip-final-runner-bridge":
            PASS4168_ADMIN_AI_ZERO_SKIP_FINAL_RUNNER_BRIDGE,
          "x-velmere-pass4169-admin-ai-receipt-freshness-expiry":
            PASS4169_ADMIN_AI_RECEIPT_FRESHNESS_EXPIRY_GATE,
          "x-velmere-pass4170-admin-ai-receipt-provenance-env-binding":
            PASS4170_ADMIN_AI_RECEIPT_PROVENANCE_ENVIRONMENT_BINDING,
          "x-velmere-pass4171-admin-ai-tamper-evident-receipt-chain":
            PASS4171_ADMIN_AI_TAMPER_EVIDENT_RECEIPT_CHAIN,
          "x-velmere-pass4172-admin-ai-release-freeze-source-manifest-seal":
            PASS4172_ADMIN_AI_RELEASE_FREEZE_DIGEST_SOURCE_MANIFEST_SEAL,
          "x-velmere-pass4173-admin-ai-post-freeze-diff-firewall-mutation-watch":
            PASS4173_ADMIN_AI_POST_FREEZE_DIFF_FIREWALL_MUTATION_WATCH,
          "x-velmere-pass4174-admin-ai-release-promotion-escrow-operator-signoff":
            PASS4174_ADMIN_AI_RELEASE_PROMOTION_ESCROW_OPERATOR_SIGNOFF_GATE,
          "x-velmere-pass4175-admin-ai-public-claim-degradation-status-board":
            PASS4175_ADMIN_AI_PUBLIC_CLAIM_DEGRADATION_STATUS_BOARD_GATE,
          "x-velmere-pass4176-admin-ai-claim-expiry-auto-downgrade-renewal":
            PASS4176_ADMIN_AI_CLAIM_EXPIRY_AUTO_DOWNGRADE_RENEWAL_GATE,
        },
      },
    );
  }

  const result = await generateGeminiAdminText({ task, locale, input });
  return securityJson(
    {
      ...result.payload,
      pass4156: buildPass4156AdminAiBoundary({
        task,
        locale,
        inputLength: input.length,
      }),
      pass4157: buildPass4157AdminAiInputRedactionFixture(input),
      pass4158,
    },
    {
      status: result.status,
      headers: {
        "x-velmere-pass4156-admin-ai-boundary":
          "admin-auth-required-input-not-echoed",
        "x-velmere-pass4157-admin-ai-input-redaction":
          "sha256-digest-only-no-raw-input",
        "x-velmere-pass4158-admin-ai-negative-fixture":
          "passed-negative-cases-digest-only",
        "x-velmere-pass4159-admin-ai-negative-replay":
          PASS4159_ADMIN_AI_NEGATIVE_CASE_REPLAY_RUNNER,
        "x-velmere-pass4160-admin-ai-signed-negative-fixture-binder":
          PASS4160_ADMIN_AI_SIGNED_NEGATIVE_FIXTURE_BINDER,
        "x-velmere-pass4161-admin-ai-final-runner-intake":
          PASS4161_ADMIN_AI_FINAL_RUNNER_INTAKE_GATE,
        "x-velmere-pass4162-admin-ai-node20-preflight-evidence":
          PASS4162_ADMIN_AI_NODE20_PREFLIGHT_EVIDENCE_BUCKET,
        "x-velmere-pass4163-admin-ai-typecheck-diagnostic-reducer":
          PASS4163_ADMIN_AI_TYPECHECK_DIAGNOSTIC_REDUCER,
      },
    },
  );
}

export async function POST(req: Request) {
  return withExpensiveRouteBudget(req, "admin_ai_post", () => handleAdminAiPost(req));
}

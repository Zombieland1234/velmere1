function shouldForceNonCryptoRealMarket(query: string) {
  const clean = query.trim().toLowerCase().replace(/\s+/g, " ");
  return /^(apple|aapl|nvidia|nvda|microsoft|msft|google|alphabet|googl|goog|tesla|tsla|amazon|amzn|meta|amd|adidas|ads|lvmh|spy|qqq|voo|s&p 500|s&p500|sp500|nasdaq|nasdaq 100|dax|vix|gold|xauusd|oil|wti|eurusd|eur\/usd)$/.test(clean);
}
